📧 BẠN CẦN EMAIL CỦA MÌNH ĐỂ NHẬN THÔNG BÁO KHI CÓ NGƯỜI ĐẶT XE

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

🎯 BƯỚC 1: ĐĂNG KÝ EMAILJS (5 phút)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

1. Truy cập: https://www.emailjs.com/
2. Click nút "Sign Up" (góc trên bên phải)
3. Điền thông tin:
   - Email: [Email của bạn - VD: your-email@gmail.com]
   - Password: [Mật khẩu tự chọn]
4. Click "Sign Up"
5. Vào email để xác nhận

✅ MIỄN PHÍ: 200 emails/tháng (đủ dùng)

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

🎯 BƯỚC 2: KẾT NỐI EMAIL (Gmail khuyến nghị - 3 phút)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

1. Sau khi login EmailJS, vào: "Email Services"
2. Click "Add New Service"
3. Chọn "Gmail" (hoặc email bạn đang dùng)
4. Click "Connect Account"
5. Đăng nhập Gmail của bạn
6. Cho phép EmailJS truy cập
7. Copy "Service ID" (VD: service_abc123)

   ┌──────────────────────────────────┐
   │ Service ID: service_abc123       │ ← Copy cái này
   └──────────────────────────────────┘

8. Lưu lại Service ID này!

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

1. Vào: "Email Templates"
2. Click "Create New Template"

🎯 BƯỚC 5: CẬP NHẬT FILE emailjs-config.js (2 phút)

1. Mở file: emailjs-config.js
2. Tìm dòng này:

   const EMAILJS_CONFIG = {
       publicKey: 'YOUR_PUBLIC_KEY',      ← Thay đổi
       serviceId: 'YOUR_SERVICE_ID',      ← Thay đổi
       templateId: 'YOUR_TEMPLATE_ID',    ← Thay đổi
       adminEmail: 'YOUR_ADMIN_EMAIL@example.com' ← Thay đổi
   };

3. Thay bằng thông tin của bạn:

   const EMAILJS_CONFIG = {
       publicKey: 'AbCdEfGhIjKlMnOp',    ← Public Key từ bước 4
       serviceId: 'service_abc123',      ← Service ID từ bước 2
       templateId: 'template_xyz789',    ← Template ID từ bước 3
       adminEmail: 'your-email@gmail.com' ← Email của bạn
   };

4. Save file (Ctrl+S)

Q: Có mất phí không?
A: MIỄN PHÍ 200 emails/tháng (không cần thẻ tín dụng)

Q: Không nhận được email?
A: Check Spam folder hoặc re-check Service/Template ID

Q: Email gửi bằng tiếng gì?
A: Tiếng Việt (đã setup sẵn trong template)

Q: Có cần backend/server không?
A: KHÔNG! EmailJS hoạt động 100% trên frontend

Q: An toàn không?
A: CÓ! EmailJS được hàng ngàn website sử dụng

🆘 NẾU GẶP VẤN ĐỀ
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

1. Mở Console (F12 trên browser)
2. Xem có lỗi màu đỏ không
3. Common errors:
   
   ❌ "emailjs is not defined"
   → Chưa load EmailJS SDK, check internet connection
   
   ❌ "Invalid publicKey"  
   → Sai Public Key, re-check emailjs-config.js
   
   ❌ "Template not found"
   → Sai Template ID, vào EmailJS Dashboard check lại
   
   ❌ "Service not found"
   → Sai Service ID, vào EmailJS Dashboard check lại
